var activeCtxEpisodeId = null;

function openActiveCtxMenu(e, episodeId) {
  e.stopPropagation();
  activeCtxEpisodeId = episodeId;
  var menu = document.getElementById('ctx-menu-active');
  menu.style.left = e.clientX + 'px';
  menu.style.top  = e.clientY + 'px';
  menu.classList.add('open');
  setTimeout(function() {
    document.addEventListener('click', closeActiveCtxMenu, { once: true });
  }, 10);
}
function closeActiveCtxMenu() {
  document.getElementById('ctx-menu-active').classList.remove('open');
}
function ctxDischarge() {
  closeActiveCtxMenu();
  if (!activeCtxEpisodeId) return;
  dischargeEpisodeId = activeCtxEpisodeId;
  document.getElementById('dc-reason').value = '';
  document.getElementById('dc-other').value  = '';
  document.getElementById('dc-other-field').style.display = 'none';
  openModal('modal-discharge');
}
function ctxViewActiveEpisode() {
  closeActiveCtxMenu();
  if (activeCtxEpisodeId) window.location.href = '/episode/' + activeCtxEpisodeId;
}

### Step 4: Add discharge/reactivate JS

- [ ] In `templates/patient.html`, find the `initDark();` call at the bottom of the `<script>` block (line ~683) and add the following functions **before** it:

```js
// ── Discharge ─────────────────────────────────────────
var dischargeEpisodeId = null;

function openDischargeModal(episodeId) {
  dischargeEpisodeId = episodeId;
  document.getElementById('dc-reason').value = '';
  document.getElementById('dc-other').value  = '';
  document.getElementById('dc-other-field').style.display = 'none';
  document.getElementById('modal-discharge').classList.add('open');
}
function closeDischargeModal() {
  document.getElementById('modal-discharge').classList.remove('open');
}
function onDcReasonChange() {
  var val = document.getElementById('dc-reason').value;
  document.getElementById('dc-other-field').style.display = val === 'Other' ? '' : 'none';
}
async function submitDischarge() {
  var reason = document.getElementById('dc-reason').value;
  if (!reason) { showToast('Please select a reason', 'err'); return; }
  if (reason === 'Other') {
    var other = document.getElementById('dc-other').value.trim();
    if (!other) { showToast('Please specify the reason', 'err'); return; }
    reason = 'Other: ' + other;
  }
  try {
    var res = await fetch('/api/episodes/' + dischargeEpisodeId + '/status', {
      method: 'PUT',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({ status: 'discharged', reason: reason })
    });
    if (!res.ok) { showToast('Failed to discharge', 'err'); return; }
    closeDischargeModal();
    showToast('Episode discharged', 'ok');
    setTimeout(function() { window.location.reload(); }, 700);
  } catch(e) { showToast('Error: ' + e.message, 'err'); }
}
async function reactivateEpisode(episodeId) {
  if (!confirm('Reactivate this episode?')) return;
  try {
    var res = await fetch('/api/episodes/' + episodeId + '/status', {
      method: 'PUT',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({ status: 'active' })
    });
    if (!res.ok) { showToast('Failed to reactivate', 'err'); return; }
    showToast('Episode reactivated', 'ok');
    setTimeout(function() { window.location.reload(); }, 700);
  } catch(e) { showToast('Error: ' + e.message, 'err'); }
}
```
